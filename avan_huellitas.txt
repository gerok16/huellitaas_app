class AuthProvider extends ChangeNotifier {
  final AuthService _service = AuthService();

  Usuario? usuario;
  bool cargando = true;

  bool get estaAutenticado => usuario != null;
  bool get esDueno => usuario?.esDueno ?? false;

  Future<void> inicializar() async {
    try {
      usuario = await _service.obtenerPerfilActual();
    } catch (_) {
      usuario = null;
    }

    cargando = false;
    notifyListeners();
  }

  Future<void> iniciarSesion(String correo, String password) async {
    cargando = true;
    notifyListeners();

    try {
      usuario = await _service.iniciarSesion(correo, password);
    } finally {
      cargando = false;
      notifyListeners();
    }
  }

  Future<void> cerrarSesion() async {
    await _service.cerrarSesion();
    usuario = null;
    notifyListeners();
  }
}

    final perfil = await _supabase

        .eq('id', usuarioAuth.id)

    if (perfil == null) {
      await _supabase.auth.signOut();
      throw Exception('No se encontrÃ³ el perfil del usuario.');
    }

    final usuario = Usuario.fromMap(perfil);

    if (!usuario.activo) {
      await _supabase.auth.signOut();
      throw Exception('Tu cuenta estÃ¡ desactivada.');
    }

    return usuario;
  }

  Future<Usuario?> obtenerPerfilActual() async {
    final actual = _supabase.auth.currentUser;
    if (actual == null) return null;

    final perfil = await _supabase
        .from('usuarios')
        .select()
        .eq('id', actual.id)
        .maybeSingle();

    if (perfil == null) return null;

    final usuario = Usuario.fromMap(perfil);
    if (!usuario.activo) {
      await cerrarSesion();
      return null;
    }

    return usuario;
  }

  }

    final datos = await _db
        .from('productos')
        .select()
        .order('nombre');

    return datos.map((p) => Producto.fromMap(p)).toList();
  }

  Future<Producto?> obtener(int id) async {
    final datos = await _db
        .from('productos')
        .select()
        .eq('id', id)
        .maybeSingle();

    if (datos == null) return null;

    return Producto.fromMap(datos);
  }

  Future<void> crear(Producto producto) async {
    await _db.from('productos').insert(producto.toMap());
  }

  Future<void> actualizar(int id, Producto producto) async {
    await _db
        .from('productos')
